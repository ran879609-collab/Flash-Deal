        function isRecommendable(cycles, globalLatestWeekStr) {

        // 可连续提报：最新期有提报 且 上一期无提报（即"最新已报"状态，排除"连报+最新已报"）
        function isConsecutiveRecommendable(cycles, globalLatestWeekStr) {
            if (cycles.length === 0) return false;

            // 最新期是否有提报
            const hasLatest = cycles.some(c => c.week === globalLatestWeekStr);
            if (!hasLatest) return false;

            // 上一期是否有提报
            const latestNum = parseInt(globalLatestWeekStr.split('-W')[1]);
            const latestYear = parseInt(globalLatestWeekStr.split('-W')[0]);
            const prevWeekStr = `${latestYear}-W${String(latestNum - 1).padStart(2, '0')}`;
            const hasPrev = cycles.some(c => c.week === prevWeekStr);

            // 只有"最新已报"（最新期有，上一期无）才返回true
            return hasLatest && !hasPrev;
        }

        function getRecommendStatus(cycles, globalLatestWeekStr) {
            if (cycles.length === 0) return { status: '⏳ 无活动', recommend: false, reason: '无活动记录', badge: 'muted' };
            const hasLatest = cycles.some(c => c.week === globalLatestWeekStr);
            let isConsecutive = false;
            if (globalLatestWeekStr) {
                const latestNum = parseInt(globalLatestWeekStr.split('-W')[1]);
                const latestYear = parseInt(globalLatestWeekStr.split('-W')[0]);
                const prevWeekStr = `${latestYear}-W${String(latestNum - 1).padStart(2, '0')}`;
                const hasPrev = cycles.some(c => c.week === prevWeekStr);
                if (hasLatest && hasPrev) isConsecutive = true;
            }
            let statusText, reasonText, badgeClass, recommend = false;
            if (hasLatest && isConsecutive) { statusText = '⚠️ 连报+最新已报'; reasonText = `最新期 ${globalLatestWeekStr} 已提报，且与上一期连续`; badgeClass = 'danger'; recommend = false; }
            else if (hasLatest) { statusText = '⏳ 最新已报'; reasonText = `最新期 ${globalLatestWeekStr} 已提报，下一期暂缓`; badgeClass = 'warning'; recommend = false; }
            else { statusText = '✅ 可提报'; reasonText = `最新期 ${globalLatestWeekStr} 无活动，可提报`; badgeClass = 'success'; recommend = true; }
            return { status: statusText, recommend, reason: reasonText, badge: badgeClass, hasLatest, isConsecutive };
        }

        function parseProductFile(file) {
            const reader = new FileReader(), ext = file.name.split('.').pop().toLowerCase();
            const parse = (data) => {
                let rows = [];
                if (ext === 'csv') {
                    const lines = data.split('\n').filter(l => l.trim());
                    if (lines.length < 2) { alert('CSV文件为空'); return; }
                    const header = lines[0].split(',').map(h => h.trim().toLowerCase());
                    const idx = { originalSku: -1, splitSku: -1, type: -1, itemId: -1 };
                    header.forEach((h, i) => {
                        if (h.includes('原始sku') || h === '原始sku') idx.originalSku = i;
                        else if (h.includes('分裂sku') || h === '分裂sku') idx.splitSku = i;
                        else if (h.includes('类型') || h === '类型') idx.type = i;
                        else if (h.includes('item id') || h.includes('itemid') || h === 'item_id') idx.itemId = i;
                    });
                    if (idx.originalSku === -1 && idx.splitSku === -1) { alert('产品表缺少「原始SKU」或「分裂SKU」列'); return; }
                    for (let i = 1; i < lines.length; i++) {
                        const cols = lines[i].split(',').map(c => c.trim());
                        if (cols.length <= Math.max(idx.originalSku, idx.splitSku, idx.type, idx.itemId)) continue;
                        const originalSku = idx.originalSku >= 0 ? cols[idx.originalSku] || '' : '';
                        const splitSku = idx.splitSku >= 0 ? cols[idx.splitSku] || '' : '';
                        const type = idx.type >= 0 ? cols[idx.type] || '' : '';
                        const itemId = idx.itemId >= 0 ? cols[idx.itemId] || '' : '';
                        const finalOriginal = originalSku || splitSku;
                        if (finalOriginal) rows.push({ originalSku: finalOriginal, splitSku, type, itemId });
                    }
                } else {
                    const workbook = XLSX.read(data, { type: 'array' });
                    const sheet = workbook.Sheets[workbook.SheetNames[0]];
                    const json = XLSX.utils.sheet_to_json(sheet);
                    json.forEach(row => {
                        const keys = Object.keys(row);
                        let originalSku = '', splitSku = '', type = '', itemId = '';
                        keys.forEach(k => {
                            const lk = k.toLowerCase();
                            if (lk.includes('原始sku') || lk === '原始sku') originalSku = String(row[k] || '');
                            else if (lk.includes('分裂sku') || lk === '分裂sku') splitSku = String(row[k] || '');
                            else if (lk.includes('类型') || lk === '类型') type = String(row[k] || '');
                            else if (lk.includes('item id') || lk.includes('itemid') || lk === 'item_id') itemId = String(row[k] || '');
                        });
                        const finalOriginal = originalSku || splitSku;
                        if (finalOriginal) rows.push({ originalSku: finalOriginal, splitSku, type, itemId });
                    });
                }
                if (rows.length === 0) { alert('未解析到有效数据'); return; }
                productData = rows; saveToStorage(); buildMerge();
            };
            if (ext === 'csv') { reader.onload = (e) => parse(e.target.result); reader.readAsText(file); }
            else { reader.onload = (e) => parse(e.target.result); reader.readAsArrayBuffer(file); }
        }

        function parseActivityFile(file) {
            const reader = new FileReader(), ext = file.name.split('.').pop().toLowerCase();
            const parse = (data) => {
                let rows = [];

                    const lines = data.split('\n').filter(l => l.trim());

                        const remark = idx.remark >= 0 ? cols[idx.remark] || '' : '';
                        if (!isNaN(year) && !isNaN(week) && sku) {
                            rows.push({ year, week: `${year}-W${String(week).padStart(2, '0')}`, sku, subId, parentId, avgOrders, sellingPrice, expectedOrders, price: isNaN(price) ? '' : price, remark });
                        }
                    }
                } else {
                    const workbook = XLSX.read(data, { type: 'array' });
                    const sheet = workbook.Sheets[workbook.SheetNames[0]];
                    const json = XLSX.utils.sheet_to_json(sheet);
                    json.forEach(row => {
                        const keys = Object.keys(row);
                        let year = NaN, week = NaN, sku = '', subId = '', parentId = '', avgOrders = '', sellingPrice = '', expectedOrders = '', price = NaN, remark = '';
                        keys.forEach(k => {
                            const lk = k.toLowerCase();
                            if (lk === '年' || lk.includes('年')) year = parseInt(row[k]);
                            else if (lk === '周' || lk.includes('周')) week = parseInt(row[k]);
                            else if (lk === 'sku' || lk.includes('sku')) sku = String(row[k] || '');
                            else if (lk.includes('子 id') || lk.includes('子id') || lk.includes('子 ID')) subId = String(row[k] || '');
                            else if (lk.includes('父 id') || lk.includes('父id') || lk.includes('父 ID')) parentId = String(row[k] || '');
                            else if (lk.includes('平均单量') || lk.includes('avg orders')) avgOrders = String(row[k] || '');
                            else if (lk.includes('售价') || lk === 'selling price') sellingPrice = String(row[k] || '');
                            else if (lk.includes('期望单量') || lk.includes('活动期望')) expectedOrders = String(row[k] || '');
                            else if (lk.includes('活动价格') || lk.includes('活动价格') || lk === '价格') price = parseFloat(row[k]);
                            else if (lk.includes('备注') || lk === 'remark') remark = String(row[k] || '');
                        });
                        if (!isNaN(year) && !isNaN(week) && sku) {
                            rows.push({ year, week: `${year}-W${String(week).padStart(2, '0')}`, sku, subId, parentId, avgOrders, sellingPrice, expectedOrders, price: isNaN(price) ? '' : price, remark });
                        }
                    });
                }
                if (rows.length === 0) { alert('未解析到有效数据'); return; }
                activityData = rows; saveToStorage(); buildMerge();
            };
            if (ext === 'csv') { reader.onload = (e) => parse(e.target.result); reader.readAsText(file); }
            else { reader.onload = (e) => parse(e.target.result); reader.readAsArrayBuffer(file); }

        function exportReportCSV(data, filename) {
            if (!data || data.length === 0) { alert('暂无数据可导出'); return; }
            let header = 'SKU,子ID,父ID,平均单量(past 90 days),售价(past 90 days),活动期望单量,活动价格,提取周期,备注\n';
            let csv = header;
            data.forEach(row => {
                csv += `"${row.sku}","${row.subId}","${row.parentId}","${row.avgOrders}","${row.sellingPrice}","${row.expectedOrders}","${row.activityPrice !== '' && !isNaN(parseFloat(row.activityPrice)) ? parseFloat(row.activityPrice).toFixed(2) : ''}","${row.lastWeek}","${row.hasRemark ? '有备注' : ''}"\n`;
            });
            const blob = new Blob(["\uFEFF" + csv], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = filename + '.csv';
            link.click();
            URL.revokeObjectURL(link.href);
        }

        function exportMainCSV() {
            if (mergedMap.size === 0) { alert('暂无数据可导出'); return; }
            const displayWeeks = getDisplayWeeks(), globalLatestWeek = getGlobalLatestWeek();
            let header = '产品类型,SKU,子ID';
            displayWeeks.forEach(w => { header += `,${w.weekStr}`; });
            header += ',连续状态,建议,原因\n';
            let csv = header;
            for (let [sku, val] of mergedMap.entries()) {
                const priceMap = new Map();
                val.cycles.forEach(c => { priceMap.set(c.week, c.price); });
                let info = globalLatestWeek ? getRecommendStatus(val.cycles, globalLatestWeek) : { status: '⏳ 无数据', recommend: false, reason: '无活动周期', badge: 'info' };
                let subId = '';
                if (val.cycles.length > 0) subId = val.cycles[val.cycles.length - 1].subId || '';
                let line = `"${val.type || ''}","${sku}","${subId}"`;
                displayWeeks.forEach(w => {
                    const price = priceMap.get(w.weekStr);
                    line += `,${price !== undefined && price !== '' && !isNaN(parseFloat(price)) ? parseFloat(price).toFixed(2) : ''}`;
                });
                line += `,"${info.status}","${info.recommend ? '建议提报' : '暂缓'}","${info.reason}"\n`;
                csv += line;
            }
            const blob = new Blob(["\uFEFF" + csv], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = '活动产品周期汇总_SKU.csv';
            link.click();
            URL.revokeObjectURL(link.href);
        }
